1-D Array Declaration.
Syntax -
------------
 int[] a = new int[5];
int -> data type
a -> Array Name.
new -> Keyword
[] -> Subscript.
5 -> Size of Array.
------------------------------
using System;
using System.Text;  // BCL /FCL

namespace DotNetBatch10AM26July
{
    class MyFirst
    {
        static void Main() // Entry Point of your program.
        {
            int[] a = new int[5];  // 1- D Array Declaration.
            Console.WriteLine("Enter 5 Elements=");
            for(int i=0;i<5;i++)
            {
                a[i] =Convert.ToInt32(Console.ReadLine());
            }
            Console.WriteLine("Print 5 Elements=");
            for (int i = 0; i < 5; i++)
            {
                Console.WriteLine(a[i]);
            }
        }
    }
}

c) Jagged Array.
-----------------------
>> It is known as 'Array of Arrays.'
>> In jagged array , we have different sizes of array and if a user want to merge to get a single array then it is known as Jagged Array.
-----------------------------------------------------------
Syntax -
-------------
int [][]a = new int[5][]; // JAGGED Array
5 -> Total number of array.
--------------------------------------------------
For Example -
-------------------
using System;
using System.Text;  // BCL /FCL

namespace DotNetBatch10AM26July
{
    class MyFirst
    {
        static void Main() // Entry Point of your program.
        {
            int [][]a = new int[5][]; // JAGGED Array
            a[0]=new int[3]{ 12,34,55};
            a[1] = new int[5] { 456, 78, 3, 23, 12 };
            a[2] = new int[2] { 987, 9 };
            a[3] = new int[1] { 8873 };
            a[4] = new int[7] { 1, 2, 3, 4, 5, 6, 7 };
            foreach (int[] arr in a)
            {
                foreach(int res in arr)
                {
                    Console.Write(res+"\t");
                }
                Console.WriteLine();
            }
        }
    }
}
